DROP TABLE IF EXISTS telemetria_sensores_campus CASCADE;

DROP TABLE IF EXISTS comentarios CASCADE;

-- ------------------------------------------------------------
-- SECCIÓN 2: DEFINICIÓN DE ESTRUCTURA DDL
-- ------------------------------------------------------------

-- 1. Tabla: estudiantes
CREATE TABLE estudiantes (
    id_estudiante BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    correo VARCHAR(150) NOT NULL UNIQUE,
    codigo_estudiantil VARCHAR(20) NOT NULL UNIQUE,
    carrera VARCHAR(100) NOT NULL,
    fecha_registro TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT chk_correo_formato CHECK (correo LIKE '%@%')
);

-- 2. Tabla: publicaciones
CREATE TABLE publicaciones (
    id_publicacion BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    id_estudiante BIGINT NOT NULL,
    contenido TEXT NOT NULL,
    fecha_publicacion TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_publicacion_estudiante 
        FOREIGN KEY (id_estudiante) 
        REFERENCES estudiantes(id_estudiante) 
        ON DELETE CASCADE
);

-- 3. Tabla: comentarios
CREATE TABLE comentarios (
    id_comentario BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    id_publicacion BIGINT NOT NULL,
    id_estudiante BIGINT NOT NULL,
    contenido TEXT NOT NULL,
    fecha_comentario TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_comentario_publicacion 
        FOREIGN KEY (id_publicacion) 
        REFERENCES publicaciones(id_publicacion) 
        ON DELETE CASCADE,
    CONSTRAINT fk_comentario_estudiante 
        FOREIGN KEY (id_estudiante) 
        REFERENCES estudiantes(id_estudiante) 
        ON DELETE CASCADE
);

-- 4. Tabla: reacciones (Relación M:N entre estudiantes y publicaciones)
CREATE TABLE reacciones (
    id_estudiante BIGINT NOT NULL,
    id_publicacion BIGINT NOT NULL,
    tipo_reaccion VARCHAR(20) NOT NULL,
    fecha_reaccion TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id_estudiante, id_publicacion),
    CONSTRAINT fk_reaccion_estudiante 
        FOREIGN KEY (id_estudiante) 
        REFERENCES estudiantes(id_estudiante) 
        ON DELETE CASCADE,
    CONSTRAINT fk_reaccion_publicacion 
        FOREIGN KEY (id_publicacion) 
        REFERENCES publicaciones(id_publicacion) 
        ON DELETE CASCADE,
    CONSTRAINT chk_tipo_reaccion 
        CHECK (tipo_reaccion IN ('me_gusta', 'me_encanta', 'interesante', 'util'))
);

-- 5. Tabla: conexiones (Red de amistades y solicitudes)
CREATE TABLE conexiones (
    id_solicitante BIGINT NOT NULL,
    id_receptor BIGINT NOT NULL,
    estado VARCHAR(20) DEFAULT 'pendiente' NOT NULL,
    fecha_solicitud TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id_solicitante, id_receptor),
    CONSTRAINT fk_conexion_solicitante 
        FOREIGN KEY (id_solicitante) 
        REFERENCES estudiantes(id_estudiante) 
        ON DELETE CASCADE,
    CONSTRAINT fk_conexion_receptor 
        FOREIGN KEY (id_receptor) 
        REFERENCES estudiantes(id_estudiante) 
        ON DELETE CASCADE,
    CONSTRAINT chk_estado_conexion 
        CHECK (estado IN ('pendiente', 'aceptada', 'rechazada', 'bloqueada')),
    CONSTRAINT chk_diferente_usuario 
        CHECK (id_solicitante <> id_receptor)
);

-- 6. Tabla IoT / Big Data: telemetria_sensores_campus
CREATE TABLE telemetria_sensores_campus (
    id_lectura BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    id_dispositivo VARCHAR(50) NOT NULL,
    temperatura REAL,
    coordenadas_gps JSONB,
    metadatos_payload JSONB,
    fecha_captura TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Indice GIN para optimización de consultas sobre payload JSONB en Big Data
CREATE INDEX idx_telemetria_payload_gin 
ON telemetria_sensores_campus USING GIN (metadatos_payload);

-- SECCIÓN 4: CONSULTAS DE VERIFICACIÓN
-- ------------------------------------------------------------

-- Consulta 1: Resumen de interacciones en publicaciones
SELECT 
    p.id_publicacion,
    e.nombre AS autor_publicacion,
    p.contenido,
    COUNT(DISTINCT c.id_comentario) AS total_comentarios,
    COUNT(DISTINCT r.id_estudiante) AS total_reacciones
FROM publicaciones p
JOIN estudiantes e ON p.id_estudiante = e.id_estudiante
LEFT JOIN comentarios c ON p.id_publicacion = c.id_publicacion
LEFT JOIN reacciones r ON p.id_publicacion = r.id_publicacion
GROUP BY p.id_publicacion, e.nombre, p.contenido;

-- Consulta 2: Extracción de atributos NoSQL JSONB con operadores de PostgreSQL
SELECT 
    id_dispositivo,
    temperatura,
    coordenadas_gps->>'lat' AS latitud,
    coordenadas_gps->>'lng' AS longitud,
    (metadatos_payload->>'humedad_pct')::INT AS humedad_porcentaje
FROM telemetria_sensores_campus
WHERE temperatura > 22.0;
